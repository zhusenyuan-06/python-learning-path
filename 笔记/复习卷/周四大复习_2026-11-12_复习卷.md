def safe_get(url, timeout=5):
    try:
        r = requests.get(url, timeout=timeout)
        r.raise_for_status()
        return (r.status_code, r.json(), None)
    except requests.exceptions.JSONDecodeError as e:
        return (None, None, f"返回的不是JSON: {e}")        # ❌ 第3位是「描述文字」
    except requests.exceptions.HTTPError as e:
        return (e.response.status_code, None, f"HTTP错误: {e}")
    except requests.RequestException as e:
        return (None, None, f"请求失败({type(e).__name__}): {e}")

# 调用方拿第3位做判断：
if 错误类型 == "JSONDecodeError":      # ❌ 拿一句描述去比「类型名」→ 永远 False
    不重试()
# 后果：JSONDecodeError 被当成「网络层错误」→ 错误地重试 3 次，
#       违反自己写的决策表（对方返回的根本不是 JSON，重试一万次也不是）

import os
DEMO = "./demo"                       # 存放待处理文件的目录
for 文件名 in os.listdir(DEMO):
    主名, 扩展名 = os.path.splitext(文件名)
    print(f"文件名 | 主名 | 扩展名")   # ❌ f 只是个标记，{} 才是插值的地方

# ⚠️ 不崩、不报警，看着像跑成功了，其实一个变量都没进去
